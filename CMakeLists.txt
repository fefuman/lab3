cmake_minimum_required(VERSION 3.16)
project(os_counter_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(os_counter src/main.cpp)
if(WIN32)
  target_compile_definitions(os_counter PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(os_counter PRIVATE Threads::Threads)
endif()
